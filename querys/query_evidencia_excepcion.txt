SELECT * FROM (
  SELECT
    'tc-sc-bi-bigdata-hdpe-pjx-dev.DFC.raw_delivered_validation_btl' AS TABLA,
    MAX(Fecha_Carga) AS FECHA_MAXIMA_CARGA,
    IF(
      MAX(Fecha_Carga) >= TIMESTAMP_SUB(CURRENT_TIMESTAMP(), INTERVAL 12 HOUR),
      'OK',
      'NOK'
    ) AS ESTADO
  FROM `tc-sc-bi-bigdata-hdpe-pjx-dev.DFC.raw_delivered_validation_btl`

  UNION ALL

  SELECT
    'tc-sc-bi-bigdata-hdpe-pjx-dev.DFC.raw_delivered_validation' AS TABLA,
    MAX(Fecha_Carga) AS FECHA_MAXIMA_CARGA,
    IF(
      MAX(Fecha_Carga) >= TIMESTAMP_SUB(CURRENT_TIMESTAMP(), INTERVAL 12 HOUR),
      'OK',
      'NOK'
    ) AS ESTADO
  FROM `tc-sc-bi-bigdata-hdpe-pjx-dev.DFC.raw_delivered_validation`
);
